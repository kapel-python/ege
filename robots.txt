# ege easy — правила обхода для поисковых роботов.
# В индексе нужны только лендинг (/) и тариф (/subscription). Приложение
# (/dashboard), админка и API — приватные зоны. Служебные страницы (/about,
# /status, /contacts, /terms, /privacy, /subscription/manage, /agent,
# /ege-result.html, /404.html, отчёт по сочинению) обходиться могут — они
# закрыты от индекса самим сервером через X-Robots-Tag: noindex, см. ниже.

User-agent: *
Allow: /$
Allow: /ege/
Allow: /assets/
Allow: /css/
Allow: /js/
Allow: /vendor/
Allow: /favicon.svg
Allow: /site.webmanifest
Allow: /llms.txt
# /main.html намеренно НЕ закрыт: это байт-в-байт копия корня, но с
# canonical на https://egeeasy.ru/ — робот обязан её прочитать, иначе
# канонический виджет не сработает и страница попадёт в индекс отдельным
# адресом. Дубликат снимает canonical, а не запрет обхода.
Disallow: /dashboard
Disallow: /index.html
Disallow: /admin
Disallow: /admin.html
# Служебные страницы (/about, /status, /contacts, /terms, /privacy,
# /subscription/manage, /agent, /404.html, /ege-result.html, /essay/*)
# намеренно НЕ закрыты здесь: сервер отдаёт их с
# X-Robots-Tag: noindex, nofollow. Закрывать их в robots.txt нельзя — робот
# не сможет прочитать noindex, и страница останется в индексе как «без
# описания». Запрет в robots.txt и noindex несовместимы.
# Файлы репозитория и служебные каталоги сервер и так не отдаёт (404),
# но явный запрет избавляет робота от попыток.
Disallow: /LICENSE
Disallow: /server/
Disallow: /test/
Disallow: /deploy/
Disallow: /backups/
Disallow: /.git/
Disallow: /api/
Disallow: /*?*

# Карта сайта отдаётся динамически (/sitemap.xml) с абсолютными URL
# (домен — из хоста запроса, переопределяется через EGE_PUBLIC_URL).
Sitemap: https://egeeasy.ru/sitemap.xml
